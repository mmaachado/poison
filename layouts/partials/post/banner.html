{{/* Banner image for a post, from the banner front matter param (featuredImage still works, for posts written before it existed). The value is looked up as a page bundle resource first, then as a file under assets/, and is otherwise used as given, which covers remote URLs and files in static/. Takes the page, or a dict with "page" and "thumb": a thumb is the short strip shown on home page cards, linked to the post. bannerInk is for single-color line art on a transparent background: the image becomes a mask filled with the text color, so the drawing follows the light/dark toggle instead of vanishing on one of the backgrounds. */}} {{- $page := . -}} {{- $thumb := false -}} {{- if reflect.IsMap . -}} {{- $page = .page -}} {{- $thumb = .thumb | default false -}} {{- end -}} {{- with $page.Params.banner | default $page.Params.featuredImage -}} {{- $src := . -}} {{- $img := or ($page.Resources.Get $src) (resources.Get $src) -}} {{- $alt := $page.Params.bannerAlt | default "" -}} {{- $ink := $page.Params.bannerInk | default false -}} {{- if and $img (eq $img.ResourceType "image") (ne $img.MediaType.SubType "svg") -}} {{- if $thumb -}} {{- $img = $img.Fill "1200x300 smart" -}} {{- else if gt $img.Width 1600 -}} {{- $img = $img.Resize "1600x" -}} {{- end -}} {{- end -}} {{- $url := $src -}} {{- with $img }}{{ $url = .RelPermalink }}{{ end -}} {{- end -}}
